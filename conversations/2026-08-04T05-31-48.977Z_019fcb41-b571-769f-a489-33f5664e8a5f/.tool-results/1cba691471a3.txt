    46    async getConfig(): Promise<Record<string, string>> {
    47      const result = await this.db.prepare('SELECT key, value FROM bot_config').all();
    48      const config: Record<string, string> = {};
    49      for (const row of result.results as any[]) {
    50        config[row.key] = row.value;
    51      }
    52      return config;
    53    }
    54  
    55    async setConfig(key: string, value: string): Promise<void> {
    56      await this.db.prepare(
    57        'INSERT INTO bot_config (key, value, updated_at) VALUES (?, ?, datetime(\'now\')) ON CONFLICT(key) DO UPDATE SET value = excluded.value, updated_at = datetime(\'now\')'
    58      ).bind(key, value).run();
    59    }

    65    async logDecision(record: DecisionRecord): Promise<number> {
    66      const result = await this.db.prepare(
    67        `INSERT INTO decisions (ticker, action, confidence, signal_source, reason, ta_data, ai_reasoning, price_at_decision, executed, execution_reason)
    68         VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`
    69      ).bind(
    70        record.ticker,
    71        record.action,
    72        record.confidence,
    73        record.signal_source,
    74        record.reason,
    75        record.ta_data,
    76        record.ai_reasoning,
    77        record.price_at_decision,
    78        record.executed,
    79        record.execution_reason
    80      ).run();
    81  
    82      return result.meta.last_row_id as number;
    83    }

   142    async upsertPosition(pos: {
   143      ticker: string;
   144      side: string;
   145      qty: number;
   146      avg_entry_price: number;
   147      current_price: number;
   148      market_value: number;
   149      unrealized_pl: number;
   150      unrealized_plpc: number;
   151      stop_loss_price: number | null;
   152      take_profit_price: number | null;
   153    }): Promise<void> {
   154      await this.db.prepare(
   155        `INSERT INTO positions (ticker, side, qty, avg_entry_price, current_price, market_value, unrealized_pl, unrealized_plpc, stop_loss_price, take_profit_price, opened_at, updated_at, closed_at, closed_pl, close_reason)
   156         VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, datetime('now'), datetime('now'), NULL, NULL, NULL)
   157         ON CONFLICT(ticker) DO UPDATE SET
   158           side = excluded.side,
   159           qty = excluded.qty,
   160           avg_entry_price = excluded.avg_entry_price,
   161           current_price = excluded.current_price,
   162           market_value = excluded.market_value,
   163           unrealized_pl = excluded.unrealized_pl,
   164           unrealized_plpc = excluded.unrealized_plpc,
   165           stop_loss_price = excluded.stop_loss_price,
   166           take_profit_price = excluded.take_profit_price,
   167           opened_at = datetime('now'),
   168           closed_at = NULL,
   169           closed_pl = NULL,
   170           close_reason = NULL,
   171           updated_at = datetime('now')`
   172      ).bind(
   173        pos.ticker,
   174        pos.side,
   175        pos.qty,
   176        pos.avg_entry_price,
   177        pos.current_price,
   178        pos.market_value,
   179        pos.unrealized_pl,
   180        pos.unrealized_plpc,
   181        pos.stop_loss_price,
   182        pos.take_profit_price
   183      ).run();
   184    }

   199    // Get tickers that were closed within the last N minutes (for re-entry cooldown)
   200    async getRecentlyClosedSymbols(cooldownMinutes: number): Promise<Set<string>> {
   201      const result = await this.db.prepare(
   202        `SELECT ticker FROM positions 
   203         WHERE closed_at IS NOT NULL 
   204         AND closed_at > datetime('now', ?)
   205         GROUP BY ticker`
   206      ).bind(`-${cooldownMinutes} minutes`).all();
   207      return new Set((result.results as any[]).map(r => r.ticker));
   208    }

   293  
   294    async getStats(): Promise<{
   295      totalDecisions: number;
   296      totalTrades: number;
   297      executedTrades: number;
   298      winRate: number;
   299      avgConfidence: number;
   300    }> {
   301      const decisionsResult = await this.db.prepare('SELECT COUNT(*) as count FROM decisions').first();
   302      const tradesResult = await this.db.prepare('SELECT COUNT(*) as count FROM trades').first();
   303      const executedResult = await this.db.prepare("SELECT COUNT(*) as count FROM trades WHERE status = 'filled'").first();
   304      const winsResult = await this.db.prepare("SELECT COUNT(*) as count FROM positions WHERE closed_pl > 0").first();
   305      const closedResult = await this.db.prepare("SELECT COUNT(*) as count FROM positions WHERE closed_at IS NOT NULL").first();
   306      const avgConfidenceResult = await this.db.prepare('SELECT AVG(confidence) as avg FROM decisions WHERE executed = 1').first();
   307  
   308      const totalTrades = (tradesResult?.count as number) || 0;
   309      const executedTrades = (executedResult?.count as number) || 0;
   310      const wins = (winsResult?.count as number) || 0;
   311      const closed = (closedResult?.count as number) || 0;
   312  
   313      return {
   314        totalDecisions: (decisionsResult?.count as number) || 0,
   315        totalTrades,
   316        executedTrades,
   317        winRate: closed > 0 ? (wins / closed) * 100 : 0,
   318        avgConfidence: (avgConfidenceResult?.avg as number) || 0,
   319      };
   320    }
