   409      function drawEquityChart(snaps) {
   410        const c = document.getElementById('equityChart'); const ctx = c.getContext('2d');
   411        const r = c.getBoundingClientRect(); c.width = r.width * devicePixelRatio; c.height = r.height * devicePixelRatio;
   412        ctx.scale(devicePixelRatio, devicePixelRatio);
   413        const w = r.width, h = r.height, p = { t: 10, r: 10, b: 30, l: 60 };
   414        const cw = w - p.l - p.r, ch = h - p.t - p.b;
   415        ctx.clearRect(0, 0, w, h);
   416        if (!snaps || snaps.length < 2) { ctx.fillStyle = '#6b7689'; ctx.font = '12px monospace'; ctx.textAlign = 'center'; ctx.fillText('Not enough data yet', w / 2, h / 2); return; }
   417        const vals = snaps.map(s => s.equity); const min = Math.min(...vals), max = Math.max(...vals);
   418        const range = max - min || 1, pr = range * 0.1, yMin = min - pr, yMax = max + pr;
   419        // Y-axis grid lines + labels
   420        ctx.strokeStyle = '#1e2638'; ctx.lineWidth = 1;
   421        for (let i = 0; i <= 4; i++) { const y = p.t + (ch / 4) * i; ctx.beginPath(); ctx.moveTo(p.l, y); ctx.lineTo(p.l + cw, y); ctx.stroke(); const v = yMax - ((yMax - yMin) / 4) * i; ctx.fillStyle = '#6b7689'; ctx.font = '10px monospace'; ctx.textAlign = 'right'; ctx.fillText('$' + v.toFixed(0), p.l - 8, y + 3); }
   422        // Equity line + fill
   423        ctx.beginPath(); vals.forEach((v, i) => { const x = p.l + (cw / (vals.length - 1)) * i; const y = p.t + ch - ((v - yMin) / (yMax - yMin)) * ch; if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y); });
   424        const up = vals[vals.length - 1] >= vals[0]; ctx.strokeStyle = up ? '#00c853' : '#ff1744'; ctx.lineWidth = 2; ctx.stroke();
   425        ctx.lineTo(p.l + cw, p.t + ch); ctx.lineTo(p.l, p.t + ch); ctx.closePath(); ctx.fillStyle = up ? 'rgba(0,200,83,0.08)' : 'rgba(255,23,68,0.08)'; ctx.fill();
   426        // Current value dot
   427        const lx = p.l + cw, ly = p.t + ch - ((vals[vals.length - 1] - yMin) / (yMax - yMin)) * ch;
   428        ctx.beginPath(); ctx.arc(lx, ly, 3, 0, Math.PI * 2); ctx.fillStyle = up ? '#00c853' : '#ff1744'; ctx.fill();
   429        // X-axis time labels (show ~5 labels spread across)
   430        ctx.fillStyle = '#6b7689'; ctx.font = '9px monospace'; ctx.textAlign = 'center';
   431        const labelCount = Math.min(5, snaps.length);
   432        for (let i = 0; i < labelCount; i++) {
   433          const idx = Math.floor((snaps.length - 1) * (i / (labelCount - 1 || 1)));
   434          const x = p.l + (cw / (vals.length - 1)) * idx;
   435          const ts = snaps[idx].timestamp;
   436          const d = new Date(ts.includes('T') ? ts + 'Z' : ts);
   437          const label = snaps.length > 200 ? d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }) : d.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' });
   438          ctx.fillText(label, x, h - 8);
   439        }
   440      }
