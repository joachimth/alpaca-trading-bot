     4  -- ============================================================
     5  -- Decisions: every AI/TA decision the bot makes
     6  -- ============================================================
     7  CREATE TABLE IF NOT EXISTS decisions (
     8    id INTEGER PRIMARY KEY AUTOINCREMENT,
     9    timestamp TEXT NOT NULL DEFAULT (datetime('now')),
    10    ticker TEXT NOT NULL,
    11    action TEXT NOT NULL CHECK(action IN ('BUY','SELL','HOLD','CLOSE')),
    12    confidence REAL NOT NULL DEFAULT 0,        -- 0.0 to 1.0
    13    signal_source TEXT NOT NULL DEFAULT 'ta',  -- 'ta', 'ai', 'ta+ai'
    14    reason TEXT,                                -- human-readable explanation
    15    ta_data TEXT,                               -- JSON: full technical analysis snapshot
    16    ai_reasoning TEXT,                          -- JSON: LLM reasoning if used
    17    price_at_decision REAL,
    18    executed INTEGER NOT NULL DEFAULT 0,        -- 0=pending, 1=executed, 2=skipped(risk), 3=failed
    19    execution_reason TEXT,                      -- why skipped/failed
    20    created_at TEXT NOT NULL DEFAULT (datetime('now'))
    21  );
    22  
    23  CREATE INDEX IF NOT EXISTS idx_decisions_timestamp ON decisions(timestamp);
    24  CREATE INDEX IF NOT EXISTS idx_decisions_ticker ON decisions(ticker);
    25  CREATE INDEX IF NOT EXISTS idx_decisions_action ON decisions(action);
    26  CREATE INDEX IF NOT EXISTS idx_decisions_executed ON decisions(executed);
    27  
    28  -- ============================================================
    29  -- Trades: executed orders
    30  -- ============================================================
    31  CREATE TABLE IF NOT EXISTS trades (
    32    id INTEGER PRIMARY KEY AUTOINCREMENT,
    33    timestamp TEXT NOT NULL DEFAULT (datetime('now')),
    34    alpaca_order_id TEXT,
    35    ticker TEXT NOT NULL,
    36    side TEXT NOT NULL CHECK(side IN ('buy','sell')),
    37    qty REAL NOT NULL,
    38    fill_price REAL,
    39    avg_fill_price REAL,
    40    status TEXT NOT NULL DEFAULT 'submitted',   -- submitted, filled, partial, canceled, rejected
    41    order_type TEXT NOT NULL DEFAULT 'market',  -- market, limit, stop, stop_limit
    42    limit_price REAL,
    43    stop_price REAL,
    44    time_in_force TEXT NOT NULL DEFAULT 'day',
    45    estimated_value REAL,
    46    decision_id INTEGER REFERENCES decisions(id),
    47    error_message TEXT,
    48    created_at TEXT NOT NULL DEFAULT (datetime('now'))
    49  );
    50  
    51  CREATE INDEX IF NOT EXISTS idx_trades_timestamp ON trades(timestamp);
    52  CREATE INDEX IF NOT EXISTS idx_trades_ticker ON trades(ticker);
    53  CREATE INDEX IF NOT EXISTS idx_trades_status ON trades(status);
    54  CREATE INDEX IF NOT EXISTS idx_trades_alpaca_id ON trades(alpaca_order_id);
    55  
    56  -- ============================================================
    57  -- Positions: current and closed positions tracked by the bot
    58  -- ============================================================
    59  CREATE TABLE IF NOT EXISTS positions (
    60    id INTEGER PRIMARY KEY AUTOINCREMENT,
    61    ticker TEXT NOT NULL UNIQUE,
    62    side TEXT NOT NULL CHECK(side IN ('long','short')),
    63    qty REAL NOT NULL,
    64    avg_entry_price REAL NOT NULL,
    65    current_price REAL,
    66    market_value REAL,
    67    unrealized_pl REAL,
    68    unrealized_plpc REAL,                       -- percentage
    69    stop_loss_price REAL,
    70    take_profit_price REAL,
    71    trailing_stop_enabled INTEGER NOT NULL DEFAULT 1,
    72    opened_at TEXT NOT NULL DEFAULT (datetime('now')),
    73    updated_at TEXT NOT NULL DEFAULT (datetime('now')),
    74    closed_at TEXT,
    75    closed_pl REAL,                             -- realized P&L when closed
    76    close_reason TEXT                           -- 'stop_loss', 'take_profit', 'signal', 'manual', 'eod'
    77  );
    78  
    79  CREATE INDEX IF NOT EXISTS idx_positions_ticker ON positions(ticker);
    80  CREATE INDEX IF NOT EXISTS idx_positions_open ON positions(closed_at);
    81  
    82  -- ============================================================
    83  -- Performance snapshots: taken each run cycle
    84  -- ============================================================
    85  CREATE TABLE IF NOT EXISTS performance_snapshots (
    86    id INTEGER PRIMARY KEY AUTOINCREMENT,
    87    timestamp TEXT NOT NULL DEFAULT (datetime('now')),
    88    account_id TEXT,
    89    equity REAL,
    90    cash REAL,
    91    buying_power REAL,
    92    portfolio_value REAL,
    93    long_market_value REAL,
    94    short_market_value REAL,
    95    positions_count INTEGER DEFAULT 0,
    96    daily_pl REAL,
    97    daily_plpc REAL,
    98    total_pl REAL,
    99    total_plpc REAL,
   100    created_at TEXT NOT NULL DEFAULT (datetime('now'))
   101  );
   102  
   103  CREATE INDEX IF NOT EXISTS idx_snapshots_timestamp ON performance_snapshots(timestamp);
   104  
   105  -- ============================================================
   106  -- Run log: every cron invocation
   107  -- ============================================================
   108  CREATE TABLE IF NOT EXISTS run_log (
   109    id INTEGER PRIMARY KEY AUTOINCREMENT,
   110    timestamp TEXT NOT NULL DEFAULT (datetime('now')),
   111    trigger TEXT NOT NULL DEFAULT 'cron',       -- 'cron', 'manual', 'api'
   112    market_open INTEGER NOT NULL DEFAULT 0,
   113    duration_ms INTEGER,
   114    decisions_made INTEGER DEFAULT 0,
   115    trades_executed INTEGER DEFAULT 0,
   116    errors INTEGER DEFAULT 0,
   117    error_details TEXT,                         -- JSON array of error messages
   118    status TEXT NOT NULL DEFAULT 'ok',          -- ok, error, skipped
   119    created_at TEXT NOT NULL DEFAULT (datetime('now'))
   120  );
   121  
   122  CREATE INDEX IF NOT EXISTS idx_runlog_timestamp ON run_log(timestamp);
   123

   127  CREATE TABLE IF NOT EXISTS bot_config (

   130    updated_at TEXT NOT NULL DEFAULT (datetime('now'))
